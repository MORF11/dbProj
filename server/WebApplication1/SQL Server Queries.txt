>----------------------------------------------------------------------------------------TABLES--------------------------------------------------------------------------------------------<

CREATE TABLE Clients (
    id INT PRIMARY KEY IDENTITY(1,1),
    clientsName NVARCHAR(50) NOT NULL,
    tariff INT NOT NULL,
    contacts NVARCHAR(50) NOT NULL
);

CREATE TABLE OrdersHistory (
    id INT PRIMARY KEY IDENTITY(1,1),
    company_id INT NOT NULL,
    deadline DATETIME2 NOT NULL,
    cost INT NOT NULL,
    areas NVARCHAR(100) NOT NULL,
    FOREIGN KEY (company_id) REFERENCES Clients (id) ON DELETE CASCADE
);

CREATE TABLE Requests (
    id INT PRIMARY KEY IDENTITY(1,1),
    stage NVARCHAR(20) NOT NULL,
    company_id INT NOT NULL,
    deadline DATETIME2  NOT NULL,
    cost INT NOT NULL,
    areas NVARCHAR(100) NOT NULL,
    FOREIGN KEY (company_id) REFERENCES Clients (id) ON DELETE CASCADE
);

CREATE TABLE Workers (
    id INT PRIMARY KEY IDENTITY(1,1),
    workersName NVARCHAR(80) NOT NULL,
    salary INT NOT NULL,
    position NVARCHAR(20) NOT NULL 
);

CREATE TABLE CancelledOrders (
    id INT PRIMARY KEY IDENTITY(1,1),
    reason NVARCHAR(200) NOT NULL,
    cost INT NOT NULL,
    cancelledordersName NVARCHAR(50) NOT NULL
);

CREATE TABLE Reviews (
    id INT PRIMARY KEY IDENTITY(1,1),
    rating INT NOT NULL,
    review NVARCHAR(500) NOT NULL,
    company_id INT NOT NULL,
    FOREIGN KEY (company_id) REFERENCES Clients (id) ON DELETE CASCADE
);
